cmake_minimum_required(VERSION 3.20)
option(MBR_BUILD_PICO "Build RP2350 firmware" OFF)
if(MBR_BUILD_PICO)
  set(PICO_BOARD pico2_w CACHE STRING "Target board")
  if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}" CACHE PATH "Pico SDK checkout")
  endif()
  include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
endif()
project(mouse_bridge_remapper C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(MBR_BUILD_PICO)
  if(NOT PICO_BOARD STREQUAL "pico2_w")
    message(FATAL_ERROR "This candidate is qualified only for Pico 2 W / RP2350")
  endif()
  pico_sdk_init()
endif()
file(GLOB_RECURSE MBR_CORE CONFIGURE_DEPENDS src/*.c)
list(FILTER MBR_CORE EXCLUDE REGEX "(_pico|/main|/qualification)\\.c$")
add_library(mbr_core STATIC ${MBR_CORE})
target_include_directories(mbr_core PUBLIC include)
target_compile_options(mbr_core PRIVATE -Wall -Wextra -Werror)
if(MBR_BUILD_PICO)
  foreach(target mouse_bridge_remapper mouse_bridge_remapper_qualification)
    add_executable(${target} src/application/main.c src/renderer/display_pico.c src/hat/hat_pico.c src/usb_hid/usb_pico.c)
    if(target STREQUAL "mouse_bridge_remapper_qualification")
      target_sources(${target} PRIVATE src/application/qualification.c)
      target_compile_definitions(${target} PRIVATE MBR_QUALIFICATION=1)
    endif()
    target_link_libraries(${target} PRIVATE mbr_core pico_stdlib hardware_spi tinyusb_device tinyusb_board)
    pico_set_program_version(${target} "0.4.0-rebuild")
    pico_enable_stdio_usb(${target} 0)
    pico_enable_stdio_uart(${target} 0)
    pico_add_extra_outputs(${target})
  endforeach()
else()
  enable_testing()
  foreach(test core ux renderer usb)
    add_executable(test_${test} tests/test_${test}.c)
    target_link_libraries(test_${test} PRIVATE mbr_core)
    target_compile_options(test_${test} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME ${test}_contract COMMAND test_${test})
  endforeach()
  add_executable(test_qualification tests/test_qualification.c src/application/qualification.c)
  target_link_libraries(test_qualification PRIVATE mbr_core)
  target_compile_options(test_qualification PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME qualification_contract COMMAND test_qualification)
  add_test(NAME architecture COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_architecture.py)
  add_test(NAME canonical_screens COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/generate_screens.py --check)
endif()
